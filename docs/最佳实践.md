### 3. API设计规范

#### ✅ 统一的API设计

```typescript
// api/device.ts
import { http } from '@jetlinks-web/utils'
import type { Device, DeviceQuery, DeviceCreateRequest, DeviceUpdateRequest } from '../types/device'

export interface PaginationResponse<T> {
  data: T[]
  total: number
  page: number
  size: number
}

export class DeviceAPI {
  private baseUrl = '/api/device'

  /**
   * 获取设备列表
   */
  async getList(query: DeviceQuery): Promise<PaginationResponse<Device>> {
    const response = await http.get<PaginationResponse<Device>>(`${this.baseUrl}/list`, {
      params: query
    })
    return response.data
  }

  /**
   * 根据ID获取设备详情
   */
  async getById(id: string): Promise<Device> {
    const response = await http.get<Device>(`${this.baseUrl}/${id}`)
    return response.data
  }

  /**
   * 创建设备
   */
  async create(data: DeviceCreateRequest): Promise<Device> {
    const response = await http.post<Device>(this.baseUrl, data)
    return response.data
  }

  /**
   * 更新设备
   */
  async update(id: string, data: DeviceUpdateRequest): Promise<Device> {
    const response = await http.put<Device>(`${this.baseUrl}/${id}`, data)
    return response.data
  }

  /**
   * 删除设备
   */
  async delete(id: string): Promise<void> {
    await http.delete(`${this.baseUrl}/${id}`)
  }

  /**
   * 批量操作
   */
  async batchDelete(ids: string[]): Promise<void> {
    await http.post(`${this.baseUrl}/batch-delete`, { ids })
  }

  /**
   * 获取设备统计信息
   */
  async getStats(): Promise<{ total: number; online: number; offline: number }> {
    const response = await http.get(`${this.baseUrl}/stats`)
    return response.data
  }
}

// 导出单例实例
export const deviceApi = new DeviceAPI()
```

### 2. 缓存策略

#### ✅ 多级缓存

```typescript
// 缓存管理器
class CacheManager {
  private componentCache = new Map<string, any>()
  private dataCache = new Map<string, { data: any; expiry: number }>()
  private readonly defaultTTL = 5 * 60 * 1000 // 5分钟

  // 组件缓存
  cacheComponent(key: string, component: any) {
    this.componentCache.set(key, component)
  }

  getComponent(key: string) {
    return this.componentCache.get(key)
  }

  // 数据缓存（带过期时间）
  cacheData(key: string, data: any, ttl = this.defaultTTL) {
    this.dataCache.set(key, {
      data,
      expiry: Date.now() + ttl
    })
  }

  getData(key: string) {
    const cached = this.dataCache.get(key)
    if (!cached) return null
    
    if (Date.now() > cached.expiry) {
      this.dataCache.delete(key)
      return null
    }
    
    return cached.data
  }

  // 清理过期缓存
  cleanup() {
    const now = Date.now()
    for (const [key, value] of this.dataCache.entries()) {
      if (now > value.expiry) {
        this.dataCache.delete(key)
      }
    }
  }

  // 清空所有缓存
  clear() {
    this.componentCache.clear()
    this.dataCache.clear()
  }
}

export const cacheManager = new CacheManager()

// 定期清理
setInterval(() => {
  cacheManager.cleanup()
}, 60000) // 每分钟清理一次
```

```vue
<!-- 懒加载组件示例 -->
<template>
  <div class="lazy-component">
    <div v-if="loading" class="loading">
      <a-spin size="large" />
    </div>
    <component

<script setup lang="ts">
import { ref, onMounted, defineAsyncComponent } from 'vue'
import { loadComponent } from '@/utils/federation-loader'

interface Props {
  moduleName: string
  componentName: string
  fallback?: any
}

const props = defineProps<Props>()

const LazyComponent = ref(null)
const loading = ref(true)
const error = ref('')

const loadAsyncComponent = async () => {
  loading.value = true
  error.value = ''
  
  try {
    const result = await loadComponent({
      remoteName: props.moduleName,
      exposedModule: `./${props.componentName}`,
      useCache: true
    })
    
    if (result.success) {
      LazyComponent.value = result.data
    } else {
      throw new Error(result.error?.message || '组件加载失败')
    }
  } catch (err) {
    error.value = err.message
    
    // 使用fallback组件
    if (props.fallback) {
      LazyComponent.value = props.fallback
      error.value = ''
    }
  } finally {
    loading.value = false
  }
}

const retry = () => {
  loadAsyncComponent()
}

onMounted(() => {
  loadAsyncComponent()
})
</script>
```

```typescript
// error-boundary.ts
import { App } from 'vue'
import { federationBridge } from '@/utils/micro-federation-bridge'

export const setupErrorBoundary = (app: App) => {
  // Vue错误处理
  app.config.errorHandler = (error, instance, info) => {
    console.error('Vue Error:', error, info)
    
    // 发送错误事件
    federationBridge.emit('error:vue', 'error-boundary', {
      error: error.message,
      info,
      component: instance?.$options.name || 'Unknown',
      timestamp: Date.now()
    })
    
    // 发送到错误监控服务
    sendErrorToService({
      type: 'vue_error',
      message: error.message,
      stack: error.stack,
      info,
      timestamp: Date.now()
    })
  }

  // 未捕获的Promise错误
  window.addEventListener('unhandledrejection', (event) => {
    console.error('Unhandled Promise Rejection:', event.reason)
    
    federationBridge.emit('error:unhandled-rejection', 'error-boundary', {
      reason: event.reason?.message || event.reason,
      timestamp: Date.now()
    })
    
    // 阻止默认的错误输出
    event.preventDefault()
  })

  // JavaScript运行时错误
  window.addEventListener('error', (event) => {
    console.error('JavaScript Error:', event.error)
    
    federationBridge.emit('error:javascript', 'error-boundary', {
      message: event.message,
      filename: event.filename,
      lineno: event.lineno,
      colno: event.colno,
      error: event.error?.message,
      timestamp: Date.now()
    })
  })
}
```

```typescript
// micro-app-error-handler.ts
import { useApplication } from '@/store'
import { federationBridge } from '@/utils/micro-federation-bridge'

export const setupMicroAppErrorRecovery = () => {
  const applicationStore = useApplication()
  
  // 监听应用加载失败
  federationBridge.on('app:preload-failed', async (event) => {
    const { appId, error } = event.payload
    
    // 尝试重新加载
    console.log(`应用 ${appId} 加载失败，尝试重新加载...`)
    
    // 等待一段时间后重试
    setTimeout(async () => {
      try {
        await applicationStore.reloadApp(appId)
        console.log(`应用 ${appId} 重新加载成功`)
        
        // 发送恢复成功事件
        federationBridge.emit('app:recovered', 'error-handler', {
          appId,
          timestamp: Date.now()
        })
      } catch (retryError) {
        console.error(`应用 ${appId} 重新加载失败:`, retryError)
        
        // 标记为永久失败
        applicationStore.setAppEnabled(appId, false)
        
        // 显示用户友好的错误信息
        window.$notification?.error({
          message: '应用加载失败',
          description: `${appId} 应用暂时无法使用，请联系管理员`,
          duration: 0 // 不自动关闭
        })
      }
    }, 2000) // 2秒后重试
  })
  
  // 监听路由加载失败
  federationBridge.on('micro-route:load-failed', (event) => {
    const { appId, error } = event.payload
```

### 3. 组件级错误处理

```vue
<!-- ErrorBoundary.vue -->
<template>
  <div class="error-boundary">
    <div v-if="hasError" class="error-container">
      <a-result
        :status="errorType"
        :title="errorTitle"
        :sub-title="errorMessage"
      >
        <template #extra>
          <a-space>
            <a-button @click="retry">重试</a-button>
            <a-button type="primary" @click="reportError">报告问题</a-button>
            <a-button @click="goHome">返回首页</a-button>
          </a-space>
        </template>
      </a-result>
    </div>
    <slot v-else />
  </div>
</template>

<script setup lang="ts">
import { ref, onErrorCaptured, computed } from 'vue'
import { useRouter } from 'vue-router'

interface Props {
  fallback?: 'warning' | 'error' | '500'
  onError?: (error: Error, info: string) => void
}

const props = withDefaults(defineProps<Props>(), {
  fallback: 'error'
})

const router = useRouter()

const hasError = ref(false)
const error = ref<Error | null>(null)
const errorInfo = ref('')

const errorType = computed(() => {
  if (!error.value) return '404'
  
  if (error.value.message.includes('网络')) return 'warning'
  if (error.value.message.includes('权限')) return '403'
  return props.fallback
})

const errorTitle = computed(() => {
  switch (errorType.value) {
    case 'warning': return '网络连接异常'
    case '403': return '无访问权限'
    case '500': return '系统异常'
    default: return '出现了一些问题'
  }
})

const errorMessage = computed(() => {
  return error.value?.message || '请稍后重试或联系管理员'
})

// 捕获子组件错误
onErrorCaptured((err, instance, info) => {
  console.error('组件错误:', err, info)
  
  hasError.value = true
  error.value = err
  errorInfo.value = info
  
  // 调用父组件错误处理
  props.onError?.(err, info)
  
  // 阻止错误继续冒泡
  return false
})

const retry = () => {
  hasError.value = false
  error.value = null
  errorInfo.value = ''
  
  // 触发组件重新渲染
  nextTick(() => {
    // 重新加载当前路由
    router.go(0)
  })
}

const reportError = () => {
  // 发送错误报告
  const report = {
    error: error.value?.message,
    stack: error.value?.stack,
    info: errorInfo.value,
    url: window.location.href,
    userAgent: navigator.userAgent,
    timestamp: new Date().toISOString()
  }
  
  // 发送到错误收集服务
  fetch('/api/error-report', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(report)
  })
  
  window.$message?.success('问题已报告，感谢您的反馈')
}

const goHome = () => {
  router.push('/')
}
</script>
```

```typescript
// shared-state.ts
import { federationBridge, useSharedData } from '@/utils/micro-federation-bridge'

// 用户状态共享
export const useSharedUserState = () => {
  const currentUser = useSharedData('currentUser', null)
  const userPermissions = useSharedData('userPermissions', [])
  
  const updateUser = (user: any) => {

      timestamp: Date.now()
    })
  }
  
  return {
    currentUser,
    userPermissions,
    updateUser,
    updatePermissions
  }
}
```

```typescript
      }, 5000)

        } else {
          reject(new Error(event.payload.error))
        }
      })
      
      // 发送请求
      this.bridge.emit('user:info-request', 'device-module', {
        userId,
        requestId: Date.now()
      })
    })
  }
  
  // 用户模块响应用户信息请求
  setupUserInfoHandler() {
    this.bridge.on('user:info-request', async (event) => {
      const { userId, requestId } = event.payload

        const userInfo = await getUserInfo(userId)
        
        this.bridge.emit('user:info-response', 'user-module', {
          requestId,
          success: true,
          data: userInfo
        })
      } catch (error) {
        this.bridge.emit('user:info-response', 'user-module', {
          requestId,
          success: false,
          error: error.message
        })
      }
    })
  }
```

```typescript
export const useResponsive = () => {

  })

    xs: 480,

  }

      width: window.innerWidth,
      height: window.innerHeight
    }

      timestamp: Date.now()
    })
  }
  
  onMounted(() => {
    window.addEventListener('resize', updateScreenSize)
  })
  
  onUnmounted(() => {
    window.removeEventListener('resize', updateScreenSize)
  })
  
  const isXS = computed(() => screenSize.value.width < breakpoints.xs)
  const isSM = computed(() => screenSize.value.width >= breakpoints.xs && screenSize.value.width < breakpoints.sm)
  const isMD = computed(() => screenSize.value.width >= breakpoints.sm && screenSize.value.width < breakpoints.md)
  const isLG = computed(() => screenSize.value.width >= breakpoints.md && screenSize.value.width < breakpoints.lg)
  const isXL = computed(() => screenSize.value.width >= breakpoints.lg)
  
  return {
    screenSize,
    breakpoints,
    isXS, isSM, isMD, isLG, isXL
  }
}
```

```typescript
// health-check.ts
export class HealthChecker {
  private checkInterval = 30000 // 30秒
  private checks = new Map<string, () => Promise<boolean>>()
  
  register(name: string, check: () => Promise<boolean>) {
    this.checks.set(name, check)
  }
  
  async runChecks(): Promise<{ [key: string]: boolean }> {
    const results: { [key: string]: boolean } = {}
    
    for (const [name, check] of this.checks) {
      try {
        results[name] = await check()
      } catch (error) {
        console.error(`健康检查失败: ${name}`, error)
        results[name] = false
      }
    }
    
    return results
  }
  
  start() {
    setInterval(async () => {
      const results = await this.runChecks()
      const failedChecks = Object.entries(results)
        .filter(([, passed]) => !passed)
        .map(([name]) => name)
      
      if (failedChecks.length > 0) {
        console.warn('健康检查失败:', failedChecks)
        
        // 发送告警
        sendAlert('health_check_failed', {
          failed: failedChecks,
          timestamp: Date.now()
        })
      }
    }, this.checkInterval)
  }
}

// 使用示例
const healthChecker = new HealthChecker()

// 注册检查项

  try {
    const response = await fetch('/api/health')
    return response.ok
  } catch {
    return false
  }
})

  const memInfo = (performance as any).memory
  if (memInfo) {
    const usageRatio = memInfo.usedJSHeapSize / memInfo.jsHeapSizeLimit
    return usageRatio < 0.9 // 内存使用率低于90%
  }
  return true
})
```
