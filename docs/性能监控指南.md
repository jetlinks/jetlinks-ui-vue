# 性能监控指南

> 微前端性能监控、分析和优化指南

## 📖 概述

本指南详细介绍微前端系统的性能监控机制，包括指标收集、性能分析、调试工具和优化策略。

## 🎛️ 阈值配置

### 1. 设置性能阈值

```typescript
// 设置全局阈值
performanceMonitor.setThresholds({
  warning: 2000,  // 2秒警告
  error: 4000,    // 4秒错误  
  timeout: 8000   // 8秒超时
})

// 不同环境使用不同阈值
const thresholds = {
  development: { warning: 5000, error: 8000, timeout: 15000 },
  testing: { warning: 3000, error: 5000, timeout: 10000 },
  production: { warning: 2000, error: 3000, timeout: 8000 }
}

performanceMonitor.setThresholds(thresholds[process.env.NODE_ENV])
```

```typescript
    // 发送性能日志

    })
```

### 3. 浏览器开发工具集成

```typescript
// 集成浏览器性能API
const integrateWithBrowserDevTools = () => {
  // 使用 Performance API 标记
  const originalStartLoading = performanceMonitor.startLoading
  const originalEndLoading = performanceMonitor.endLoading
  
  performanceMonitor.startLoading = function(name: string) {
    // 添加Performance标记
    if (typeof performance !== 'undefined' && performance.mark) {
      performance.mark(`${name}-start`)
    }
    
    return originalStartLoading.call(this, name)
  }
  
  performanceMonitor.endLoading = function(id: string, resourceSize?: number) {
    const metrics = this.metrics.get(id)
    if (metrics && typeof performance !== 'undefined') {
      // 添加结束标记
      if (performance.mark) {
        performance.mark(`${metrics.name}-end`)
      }
      
      // 测量持续时间
      if (performance.measure) {
        performance.measure(
          `${metrics.name}-duration`,
          `${metrics.name}-start`,
          `${metrics.name}-end`
        )
      }
    }
    
    return originalEndLoading.call(this, id, resourceSize)
  }
}

// 在开发环境启用
if (process.env.NODE_ENV === 'development') {
  integrateWithBrowserDevTools()
}
```

## 📋 监控清单

### 1. 应用级监控

```typescript
// 应用加载监控
import { useApplication } from '@/store'

const monitorApplicationLoading = () => {
  const app = useApplication()
  
  // 监控应用预加载
  const originalPreloadApp = app.preloadApp
  app.preloadApp = async function(appId: string) {
    const monitorId = performanceMonitor.startLoading(`app-preload-${appId}`)
    
    try {
      const result = await originalPreloadApp.call(this, appId)
      performanceMonitor.endLoading(monitorId)
      return result
    } catch (error) {
      performanceMonitor.endLoadingWithError(monitorId, error)
      throw error
    }
  }
}
```

```typescript
const monitorComponentLoading = () => {

    }
  }
}
```

### 3. 路由级监控

```typescript
// 路由切换监控
import { router } from '@/router'

const monitorRouteChanges = () => {
  let routeStartTime: number
  
  router.beforeEach((to, from, next) => {
    routeStartTime = performance.now()
    
    // 开始路由监控
    if (to.path !== from.path) {
      performanceMonitor.startLoading(`route-${from.path}-to-${to.path}`)
    }
    
    next()
  })
  
  router.afterEach((to, from) => {
    const routeEndTime = performance.now()
    const duration = routeEndTime - routeStartTime
    
    // 记录路由切换时间
    console.log(`路由切换耗时: ${from.path} -> ${to.path} (${duration.toFixed(2)}ms)`)
    
    // 发送到监控系统
    if (duration > 100) { // 超过100ms记录
      sendRouteMetrics({
        from: from.path,
        to: to.path,
        duration,
        timestamp: Date.now()
      })
    }
  })
}
```

## 📈 性能优化

### 1. 基于监控数据的优化

```typescript
// 分析性能瓶颈
const analyzePerformanceBottlenecks = () => {
  const stats = performanceMonitor.getStats()
  const allMetrics = performanceMonitor.getAllMetrics()
  
  // 找出慢加载的应用
  const slowApps = allMetrics
    .filter(m => m.loadTime > 2000)
    .sort((a, b) => b.loadTime - a.loadTime)
  
  console.log('慢加载应用:', slowApps)
  
  // 找出失败率高的应用
  const failedApps = allMetrics
    .filter(m => m.status === 'error')
    .reduce((acc, m) => {
      acc[m.name] = (acc[m.name] || 0) + 1
      return acc
    }, {})
  
  console.log('失败率统计:', failedApps)
  
  // 生成优化建议
  const suggestions = generateOptimizationSuggestions(stats, slowApps, failedApps)
  console.log('优化建议:', suggestions)
}

const generateOptimizationSuggestions = (stats, slowApps, failedApps) => {
  const suggestions = []
  
  if (stats.averageLoadTime > 2000) {
    suggestions.push('平均加载时间过长，建议启用预加载策略')
  }
  
  if (stats.successRate < 95) {
    suggestions.push('成功率偏低，检查网络连接和服务可用性')
  }
  
  if (slowApps.length > 0) {
    suggestions.push(`以下应用加载较慢: ${slowApps.map(a => a.name).join(', ')}`)
  }
  
  return suggestions
}
```

### 2. 自动优化策略

```typescript
// 基于性能数据自动调整策略
const autoOptimize = () => {
  const stats = performanceMonitor.getStats()
  
  // 如果平均加载时间过长，启用预加载
  if (stats.averageLoadTime > 3000) {
    import('@/utils/federation-preloader').then(({ preloader }) => {
      // 对常用应用启用预加载
      const frequentApps = getFrequentlyUsedApps()
      frequentApps.forEach(appId => {
        preloader.addTask(appId, {
          strategy: 'idle',
          priority: 8
        })
      })
    })
  }
  
  // 如果失败率过高，增加重试机制
  if (stats.successRate < 90) {
    import('@/utils/federation-loader').then(({ FederationLoader }) => {
      // 启用自动重试
      FederationLoader.enableAutoRetry(3) // 最多重试3次
    })
  }
}

// 定期执行自动优化
setInterval(autoOptimize, 60000) // 每分钟检查一次
```

## ⚠️ 最佳实践

- 选择合适的监控粒度，避免过度监控

- 结合多个维度分析性能问题
- 关注趋势变化而非单次数据
- 建立性能基线进行对比

### 3. 优化措施
- 基于数据做优化决策
- 渐进式优化，避免过度优化
- 验证优化效果

### 4. 告警机制
- 设置分级告警
- 避免告警疲劳
