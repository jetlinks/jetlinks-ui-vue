# 模块注册中心

> 统一的模块资源管理和注册系统

## 📖 概述

模块注册中心是微前端架构的核心组件，负责统一管理各个子模块的组件、API、工具函数、路由等资源。支持本地模块、联邦模块和微应用模块的统一注册和管理。

## 🏗️ 架构设计

### 核心组件

```typescript
ModuleRegistry              // 模块注册中心  
├── ModuleResource         // 模块资源接口
├── ModuleMetadata         // 模块元数据
├── ModuleStatus          // 模块状态枚举
└── RegisterOptions       // 注册选项
```

### 资源类型

```typescript
interface ModuleResource {
  moduleId: string                    // 模块ID
  components?: Record<string, any>    // Vue组件
  apis?: Record<string, any>         // API接口
  utils?: Record<string, any>        // 工具函数
  stores?: Record<string, any>       // 状态管理
  routes?: RouteRecordRaw[]          // 路由配置
  [key: string]: any                 // 其他资源
}
```

## 🚀 快速开始

### 1. 基础使用

```typescript
import { moduleRegistry } from '@/utils/module-registry'

// 注册模块
moduleRegistry.register('device-manager', {
  components: {
    DeviceList: () => import('./components/DeviceList.vue'),
    DeviceDetail: () => import('./components/DeviceDetail.vue')
  },
  apis: {
    deviceApi: {
      getList: () => api.get('/devices'),
      getById: (id) => api.get(`/devices/${id}`)
    }
  },
  utils: {
    deviceUtils: {
      formatStatus: (status) => statusMap[status],
      validateDevice: (device) => device.name && device.type
    }
  }
})

// 获取模块资源
const deviceModule = moduleRegistry.getModule('device-manager')
console.log('设备模块:', deviceModule)

// 获取特定资源类型
const deviceComponents = moduleRegistry.getResource('device-manager', 'components')
```


## 📦 模块注册

### 1. 本地模块注册

```typescript
// src/modules/device-manager-ui/index.ts
import DeviceList from './components/DeviceList.vue'
import DeviceDetail from './components/DeviceDetail.vue'
import { deviceApi } from './api'
import { useDeviceStore } from './store'
import { routes } from './routes'

// 导出给模块加载器
export default {
  // 获取默认路由
  getDefaultRoutes() {
    return routes
  },
  
  // 获取过滤路由 (不需要token的路由)
  getFilterRoutes() {
    return ['/device/public']
  },
  
  // 注册模块资源
  register() {
    const { moduleRegistry } = import('@/utils/module-registry')
    
    moduleRegistry.register('device-manager', {
      components: {
        DeviceList,
        DeviceDetail
      },
      apis: {
        deviceApi
      },
      stores: {
        useDeviceStore
      },
      routes: routes,
      utils: {
        deviceFormatter: {
          formatStatus: (status) => statusMap[status]
        }
      }
    })
  }
}
```

## 🔄 生命周期管理

### 1. 模块加载生命周期

```typescript
import { federationBridge } from '@/utils/micro-federation-bridge'

// 监听模块注册事件
federationBridge.on('module:registered', (event) => {
  const { moduleId, resource } = event.payload
  console.log(`模块 ${moduleId} 已注册`, resource)
})

// 监听联邦模块加载完成
federationBridge.on('federation-module:loaded', (event) => {
  const { moduleId, remoteName } = event.payload
  console.log(`联邦模块 ${moduleId} 加载完成 (${remoteName})`)
})

// 监听模块加载失败
federationBridge.on('federation-module:load-failed', (event) => {
  const { moduleId, error } = event.payload
  console.error(`联邦模块 ${moduleId} 加载失败:`, error)
})

// 监听模块卸载
federationBridge.on('module:unloaded', (event) => {
  const { moduleId } = event.payload
  console.log(`模块 ${moduleId} 已卸载`)
})
```

```typescript
// 根据条件注册不同资源
const registerModule = (moduleId: string, env: string) => {
  const baseResource = {
    components: { /* 基础组件 */ },

  }
```

### 3. 依赖管理

```typescript
// 注册带依赖的模块
moduleRegistry.register('advanced-device-manager', {
  components: { AdvancedDeviceList },
  dependencies: ['device-manager', 'user-manager'] // 声明依赖
})

// 检查依赖是否满足
const checkDependencies = (moduleId: string) => {
  const metadata = moduleRegistry.getMetadata(moduleId)
  if (metadata?.dependencies) {
    const missingDeps = metadata.dependencies.filter(dep => 
      !moduleRegistry.hasModule(dep)
    )
    
    if (missingDeps.length > 0) {
      console.warn(`模块 ${moduleId} 缺少依赖:`, missingDeps)
      return false
    }
  }
  return true
}
```

## 🎯 实际应用

### 1. 在Vue组件中使用

```vue
<template>
  <div>
    <component :is="DeviceList" v-if="DeviceList" />
    <div v-else>设备列表组件加载中...</div>
  </div>
</template>

<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { getResourceItem, hasResourceItem } from '@/utils/module-registry'

const DeviceList = ref(null)

onMounted(() => {
  // 检查组件是否已注册
  if (hasResourceItem('device-manager', 'components', 'DeviceList')) {
    DeviceList.value = getResourceItem('device-manager', 'components', 'DeviceList')
  } else {
    // 监听模块注册事件
    const unsubscribe = federationBridge.on('module:registered', (event) => {
      const { moduleId } = event.payload
      if (moduleId === 'device-manager') {
        DeviceList.value = getResourceItem('device-manager', 'components', 'DeviceList')
        unsubscribe() // 取消监听
      }
    })
  }
})
</script>
```

```typescript
// 检查模块完整性
const validateModule = (moduleId: string) => {
  const module = getModule(moduleId)
  if (!module) {
    console.error(`模块 ${moduleId} 未注册`)
    return false
  }
  
  const metadata = moduleRegistry.getMetadata(moduleId)
  if (metadata?.status === ModuleStatus.ERROR) {
    console.error(`模块 ${moduleId} 状态异常:`, metadata.error)
    return false
  }
  
  return true
}

// 检查资源引用
const validateResourceReferences = () => {
  const allModules = moduleRegistry.getAllModules()
  
  allModules.forEach((module, moduleId) => {
    if (module.components) {
      Object.keys(module.components).forEach(componentName => {
        if (!module.components[componentName]) {
          console.warn(`模块 ${moduleId} 的组件 ${componentName} 为空`)
        }
      })
    }
  })
}
```

### 3. 性能监控

```typescript
// 监控模块注册性能
import { performanceMonitor } from '@/utils/federation-performance-monitor'

const originalRegister = moduleRegistry.register
moduleRegistry.register = function(moduleId, resource, options) {
  const monitorId = performanceMonitor.startLoading(`module_register_${moduleId}`)
  
  try {
    const result = originalRegister.call(this, moduleId, resource, options)
    performanceMonitor.endLoading(monitorId)
    return result
  } catch (error) {
    performanceMonitor.endLoadingWithError(monitorId, error)
    throw error
  }
}
```
